# ZIP 更新器继续使用原项目附带的 zlib 静态源码。
set(ZLIB_DIR "${CMAKE_SOURCE_DIR}/vendor/libpng/packages/zlib.static.1.2.5/build/native/src")
set(ZLIB_HEADERS "${CMAKE_SOURCE_DIR}/vendor/libpng/packages/zlib.static.1.2.5/build/native/include")
file(GLOB ZLIB_SOURCES "${ZLIB_DIR}/*.c")
add_library(zlib STATIC ${ZLIB_SOURCES})
target_include_directories(zlib PUBLIC "${ZLIB_DIR}" "${ZLIB_HEADERS}")

add_library(download-core STATIC src/core_text.cpp src/process.cpp src/about.cpp src/download_core.cpp src/download_application.cpp
    src/task_model.cpp src/task_queue.cpp src/download_scheduler.cpp src/queue_persistence.cpp src/collection_model.cpp src/output_model.cpp src/download_preferences.cpp src/download_capabilities.cpp src/application_settings.cpp src/naming_model.cpp src/i18n.cpp src/queue_completion.cpp src/queue_groups.cpp src/components.cpp src/component_probe.cpp src/component_application.cpp src/component_update.cpp src/component_update_native.cpp src/component_update_application.cpp)
target_link_libraries(download-core PUBLIC crypt32 winhttp version bcrypt advapi32 shell32 PRIVATE zlib)
target_include_directories(download-core PUBLIC src PRIVATE "${ZLIB_HEADERS}" "${CMAKE_BINARY_DIR}/generated")
target_compile_definitions(download-core PRIVATE UNICODE _UNICODE NOMINMAX)
target_compile_options(download-core PRIVATE /utf-8 /EHsc)
if(EXISTS "${CMAKE_SOURCE_DIR}/tests/core_smoke.cpp")
add_executable(core-smoke EXCLUDE_FROM_ALL tests/core_smoke.cpp)

set_property(TARGET core-smoke APPEND PROPERTY LINK_DEPENDS "${CMAKE_SOURCE_DIR}/src/manifest.xml")
add_executable(queue-model-smoke EXCLUDE_FROM_ALL tests/queue_model.cpp)

endif()

# About 的构建版本读取实际头文件、依赖配置与固定 SDK 元数据。
execute_process(COMMAND git describe --always --dirty WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}"
    OUTPUT_VARIABLE ABOUT_REVISION OUTPUT_STRIP_TRAILING_WHITESPACE ERROR_QUIET)
file(STRINGS "${CMAKE_SOURCE_DIR}/ui/lucide.min.js" ABOUT_LINE LIMIT_COUNT 6 REGEX "@license lucide v")
string(REGEX MATCH "[0-9]+\\.[0-9]+\\.[0-9]+" ABOUT_LUCIDE "${ABOUT_LINE}")
set(ABOUT_WEBVIEW_SDK "")
if(EXISTS "${WEBVIEW2_SDK_DIR}/Microsoft.Web.WebView2.nuspec")
    file(STRINGS "${WEBVIEW2_SDK_DIR}/Microsoft.Web.WebView2.nuspec" ABOUT_LINE REGEX "<version>")
    string(REGEX MATCH "[0-9]+\\.[0-9]+\\.[0-9]+\\.[0-9]+" ABOUT_WEBVIEW_SDK "${ABOUT_LINE}")
endif()
file(READ "${CMAKE_SOURCE_DIR}/LICENSE" ABOUT_PROJECT_LICENSE)
file(READ "${CMAKE_SOURCE_DIR}/ui/lucide-LICENSE" ABOUT_LUCIDE_LICENSE)
configure_file(src/build_info.hpp.in generated/build_info.hpp @ONLY)
